# 3. READ OBSERVED FLIGHT COORDINATES

def read_track(path):

    if not path.exists():
        raise FileNotFoundError(
            f"Missing observation file: {path}"
        )

    df = pd.read_csv(path, low_memory=False)

    required = {"G_LAT", "G_LONG"}

    if not required.issubset(df.columns):
        raise ValueError(
            f"Missing coordinate columns: {path}"
        )

    lat = pd.to_numeric(
        df["G_LAT"], errors="coerce"
    )

    lon = pd.to_numeric(
        df["G_LONG"], errors="coerce"
    )

    valid = (
        lat.notna()
        & lon.notna()
        & lat.between(-90, 90)
        & lon.between(-180, 360)
    )

    track = pd.DataFrame({
        "latitude": lat[valid].to_numpy(),
        "longitude": lon[valid].to_numpy()
    })

    # Normalize to -180 through +180 degrees.
    track["longitude"] = (
        (track["longitude"] + 180) % 360
    ) - 180

    if len(track) < 2:
        raise ValueError(
            f"Insufficient valid coordinates: {path}"
        )

    return track

from pathlib import Path
import pandas as pd
import re

ROOT = Path(
    r"C:\Users\chris\OneDrive\Desktop\Publication Projects"
    r"\Eathdata\ATTREX"
    r"\ATTREX-Aircraft_navigational_and_meteorological_Measurements_1-20261025_192825"
)

TARGET_SEGMENTS = [1450, 1986]

U_NAMES = [
    "u_correlation", "corr_u",
    "u_corr", "correlation_u"
]

V_NAMES = [
    "v_correlation", "corr_v",
    "v_corr", "correlation_v"
]

SEGMENT_NAMES = ["segment", "segment_id"]

results = []

print("\nSearching existing validation CSV files...\n")

for path in ROOT.rglob("*.csv"):

    # Avoid large observation, trial, and route files.
    name = path.name.lower()

    if not any(term in name for term in [
        "metric", "validation", "summary",
        "comparison", "collocat", "matched"
    ]):
        continue

    try:
        df = pd.read_csv(path, nrows=100)
    except Exception:
        continue

    u_col = next(
        (c for c in U_NAMES if c in df.columns), None
    )

    v_col = next(
        (c for c in V_NAMES if c in df.columns), None
    )

    if u_col is None or v_col is None:
        continue

    segment_col = next(
        (c for c in SEGMENT_NAMES if c in df.columns),
        None
    )

    if segment_col is not None:

        for segment in TARGET_SEGMENTS:

            subset = df[
                pd.to_numeric(
                    df[segment_col], errors="coerce"
                ) == segment
            ]

            for _, row in subset.iterrows():

                results.append({
                    "segment": segment,
                    "u_correlation": row[u_col],
                    "v_correlation": row[v_col],
                    "source": str(path)
                })

    else:
        # Only assign segment from an explicit filename match.
        for segment in TARGET_SEGMENTS:

            if re.search(
                rf"(?<!\d){segment}(?!\d)",
                path.name
            ):

                for _, row in df.iterrows():

                    results.append({
                        "segment": segment,
                        "u_correlation": row[u_col],
                        "v_correlation": row[v_col],
                        "source": str(path)
                    })

if results:

    output = pd.DataFrame(results).drop_duplicates()

    print("FOUND CORRELATION RESULTS:\n")
    print(output.to_string(index=False))

else:

    print(
        "No stored correlation metrics found "
        "for segments 1450 and 1986."
    )
